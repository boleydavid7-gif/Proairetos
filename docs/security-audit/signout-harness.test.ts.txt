// @vitest-environment jsdom
// SCRATCH AUDIT HARNESS (not part of the repo): real syncController, real supabase-js, stub network.
import 'fake-indexeddb/auto';
import { describe, expect, it, vi, beforeAll } from 'vitest';

vi.stubEnv('VITE_SUPABASE_URL', 'https://stub.test');
vi.stubEnv('VITE_SUPABASE_ANON_KEY', 'anon');

const net = vi.hoisted(() => ({ calls: [] as string[], online: true, signedOutOnServer: false }));
const b64 = (o: unknown) => btoa(JSON.stringify(o)).replace(/=+$/, '').replace(/\+/g, '-').replace(/\//g, '_');
const jwt = () => `${b64({ alg: 'HS256' })}.${b64({ sub: 'u1', exp: Math.floor(Date.now() / 1000) + 3600, role: 'authenticated' })}.sig`;
const user = { id: 'u1', email: 'a@b.test', aud: 'authenticated', app_metadata: {}, user_metadata: {}, created_at: '2026-01-01T00:00:00Z' };

beforeAll(() => {
  globalThis.fetch = (async (input: RequestInfo | URL, init?: RequestInit) => {
    const url = String(input instanceof Request ? input.url : input);
    const method = init?.method ?? 'GET';
    net.calls.push(`${method} ${url.replace('https://stub.test', '')}`);
    if (!net.online) throw new TypeError('Failed to fetch');
    const ok = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { 'content-type': 'application/json' } });
    if (url.includes('/auth/v1/logout')) { net.signedOutOnServer = true; return new Response(null, { status: 204 }); }
    if (url.includes('/auth/v1/user')) return ok(user);
    if (url.includes('grant_type=refresh_token')) return net.signedOutOnServer ? ok({ error: 'invalid_grant' }, 400) : ok({ access_token: jwt(), refresh_token: 'r2', expires_in: 3600, token_type: 'bearer', user });
    if (url.includes('/rest/v1/')) return ok([]);
    return ok({});
  }) as typeof fetch;
});

async function dump() {
  const ls: Record<string, number> = {};
  for (let i = 0; i < localStorage.length; i++) { const k = localStorage.key(i)!; ls[k] = (localStorage.getItem(k) ?? '').length; }
  const idb: Record<string, Record<string, number>> = {};
  for (const info of await indexedDB.databases()) {
    const db: IDBDatabase = await new Promise((res, rej) => { const r = indexedDB.open(info.name!); r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
    idb[info.name!] = {};
    for (const name of Array.from(db.objectStoreNames)) {
      idb[info.name!][name] = await new Promise<number>((res) => { const t = db.transaction(name).objectStore(name).count(); t.onsuccess = () => res(t.result); });
    }
    db.close();
  }
  return { ls, idb };
}

describe('audit: what a sign-out leaves, and what another tab keeps', () => {
  it('measures', async () => {
    const { deviceDatabase } = await import('../../app/services');
    const { createIndexedDbSyncStateStore } = await import('../../data/sync/localStores');
    const db = (await deviceDatabase)!;
    const st = createIndexedDbSyncStateStore(Promise.resolve(db));
    const key = await crypto.subtle.generateKey({ name: 'AES-GCM', length: 256 }, false, ['encrypt', 'decrypt']);
    await st.setMeta('dataKey', key);
    await st.setMeta('handoffKey', 'HANDOFF-SECRET');
    await st.setMeta('lastSync', '2026-11-06T10:00:00Z');
    // Session as supabase-js keeps it, and plaintext the person owns.
    localStorage.setItem('proairetos.auth', JSON.stringify({ access_token: jwt(), refresh_token: 'r1', expires_at: Math.floor(Date.now() / 1000) + 3600, expires_in: 3600, token_type: 'bearer', user }));
    localStorage.setItem('proairetosAccount', 'u1');
    localStorage.setItem('proairetos.displayName', 'Sam PRIVATE-NAME');
    localStorage.setItem('therapeia:medicine:m1', JSON.stringify({ name: 'PRIVATE-MED', dose: '50 mg' }));
    localStorage.setItem('askesis:strava', JSON.stringify({ access_token: 'STRAVA-ACCESS', refresh_token: 'STRAVA-REFRESH' }));
    localStorage.setItem('proairetos.captureInbox', JSON.stringify({ token: 'CAPTURE-TOKEN', privateKey: 'PRIV' }));
    localStorage.setItem('proairetos.calendarFeed', JSON.stringify({ enabled: true, options: { shifts: true } }));
    await new Promise<void>((resolve) => { const tx = db.transaction('lifeItems', 'readwrite'); tx.objectStore('lifeItems').put({ id: 'i1', title: 'PRIVATE-ITEM', status: 'OPEN', type: 'TASK', createdAt: 'x', updatedAt: 'x' }); tx.oncomplete = () => resolve(); });

    const A = await import('../../app/sync/syncController');
    await A.startSync();
    console.log('A after start:', A.syncStatus.get().phase);
    const sealed = await A.sealWithAccount('audit', new TextEncoder().encode('SECRET-BYTES').buffer as ArrayBuffer);
    expect(sealed).not.toBeNull();

    vi.resetModules();
    const B = await import('../../app/sync/syncController');
    await B.startSync();
    console.log('B after start:', B.syncStatus.get().phase);

    const before = await dump();
    await A.signOut();
    await new Promise((r) => setTimeout(r, 400));
    const after = await dump();
    console.log('A phase:', A.syncStatus.get().phase, '| B phase:', B.syncStatus.get().phase);
    console.log('network:', net.calls.filter((c) => !c.includes('/rest/v1/account_devices')).join(' | '));
    console.log('localStorage keys removed by sign-out:', Object.keys(before.ls).filter((k) => !(k in after.ls)));
    console.log('localStorage keys kept:', Object.keys(after.ls));
    console.log('IDB before:', JSON.stringify(before.idb));
    console.log('IDB after :', JSON.stringify(after.idb));
    const opened = await B.openWithAccount('audit', sealed!.sealed.buffer as ArrayBuffer).then(() => 'B CAN STILL OPEN SEALED BYTES', () => 'B opens nothing');
    console.log('other tab (B) after sign-out in A:', opened);
    const reseal = await B.sealWithAccount('audit', new TextEncoder().encode('x').buffer as ArrayBuffer);
    console.log('other tab (B) can seal new bytes:', reseal !== null);
  });

  async function seedSignedIn() {
    const { deviceDatabase } = await import('../../app/services');
    const { createIndexedDbSyncStateStore } = await import('../../data/sync/localStores');
    const db = (await deviceDatabase)!;
    const st = createIndexedDbSyncStateStore(Promise.resolve(db));
    await st.clear();
    await st.setMeta('dataKey', await crypto.subtle.generateKey({ name: 'AES-GCM', length: 256 }, false, ['encrypt', 'decrypt']));
    localStorage.setItem('proairetos.auth', JSON.stringify({ access_token: jwt(), refresh_token: 'r1', expires_at: Math.floor(Date.now() / 1000) + 3600, expires_in: 3600, token_type: 'bearer', user }));
    localStorage.setItem('proairetosAccount', 'u1');
    return { db, st };
  }

  it('signing out with no connection', async () => {
    vi.resetModules();
    net.calls.length = 0; net.signedOutOnServer = false; net.online = true;
    await seedSignedIn();
    const A = await import('../../app/sync/syncController');
    await A.startSync();
    net.online = false;
    await A.signOut();
    net.online = true;
    const after = await dump();
    console.log('OFFLINE sign-out: phase', A.syncStatus.get().phase, '| session in storage:', 'proairetos.auth' in after.ls, '| syncMeta rows:', after.idb.proairetos.syncMeta, '| server logout reached:', net.signedOutOnServer);
  });

  it('a session ended from elsewhere (revoked on another device, or a refresh that fails): is the key kept?', async () => {
    vi.resetModules();
    net.calls.length = 0; net.signedOutOnServer = false; net.online = true;
    await seedSignedIn();
    const A = await import('../../app/sync/syncController');
    await A.startSync();
    console.log('B-device before: phase', A.syncStatus.get().phase);
    // Another device signs out everywhere: the server drops the refresh token; this device's access token is old.
    net.signedOutOnServer = true;
    localStorage.setItem('proairetos.auth', JSON.stringify({ access_token: jwt(), refresh_token: 'r1', expires_at: Math.floor(Date.now() / 1000) - 10, expires_in: 3600, token_type: 'bearer', user }));
    const { supabase } = await import('../../data/sync/supabase');
    const client = await supabase();
    const got = await client.auth.getSession();
    await new Promise((r) => setTimeout(r, 400));
    const mid = await dump();
    console.log('REVOKED elsewhere: session now', got.data.session ? 'present' : 'gone', '| phase', A.syncStatus.get().phase, '| session key in storage:', 'proairetos.auth' in mid.ls, '| syncMeta rows (the data key):', mid.idb.proairetos.syncMeta);
    // Someone signs in to the same account here again: is the passphrase or passkey asked for?
    net.signedOutOnServer = false;
    await client.auth.setSession({ access_token: jwt(), refresh_token: 'r2' });
    await new Promise((r) => setTimeout(r, 600));
    console.log('SIGN-IN again (same account), no passphrase or passkey given: phase', A.syncStatus.get().phase);
  });
});
